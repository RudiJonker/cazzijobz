Step 1: Create the Welcome Screen
Create /src/screens/WelcomeScreen.js:

javascript
import React from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';
import { lightTheme } from '../styles/theme';

export default function WelcomeScreen({ navigation }) {
  return (
    <View style={[styles.container, { backgroundColor: lightTheme.colors.background }]}>
      <Text style={[styles.title, { color: lightTheme.colors.text }]}>
        Welcome to cazzijobz
      </Text>
      <Text style={[styles.subtitle, { color: lightTheme.colors.text }]}>
        Choose how you'll use the app
      </Text>
      
      <View style={styles.roleContainer}>
        <TouchableOpacity 
          style={[styles.roleCard, { backgroundColor: lightTheme.colors.card }]}
          onPress={() => navigation.navigate('Terms', { role: 'worker' })}
        >
          <Text style={styles.roleIcon}>👷</Text>
          <Text style={[styles.roleTitle, { color: lightTheme.colors.text }]}>
            Looking for Work
          </Text>
          <Text style={[styles.roleDescription, { color: lightTheme.colors.text }]}>
            Find casual jobs in your area
          </Text>
        </TouchableOpacity>

        <TouchableOpacity 
          style={[styles.roleCard, { backgroundColor: lightTheme.colors.card }]}
          onPress={() => navigation.navigate('Terms', { role: 'employer' })}
        >
          <Text style={styles.roleIcon}>💼</Text>
          <Text style={[styles.roleTitle, { color: lightTheme.colors.text }]}>
            Hiring Workers
          </Text>
          <Text style={[styles.roleDescription, { color: lightTheme.colors.text }]}>
            Post jobs and find reliable help
          </Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}

Step 2: Update App.js for Navigation
Replace your current App.js:

javascript
import React from 'react';
import { NavigationContainer } from '@react-navigation/native';
import { createStackNavigator } from '@react-navigation/stack';
import WelcomeScreen from './src/screens/WelcomeScreen';
import { lightTheme } from './src/styles/theme';

const Stack = createStackNavigator();

export default function App() {
  return (
    <NavigationContainer theme={{ colors: { background: lightTheme.colors.background } }}>
      <Stack.Navigator screenOptions={{ headerShown: false }}>
        <Stack.Screen name="Welcome" component={WelcomeScreen} />
        {/* We'll add TermsScreen next */}
      </Stack.Navigator>
    </NavigationContainer>
  );
}
Step 3: Test the Navigation
Run npx expo start --clear and you should see the Welcome screen with two role cards. Tapping them will give an error (because TermsScreen doesn't exist yet), but the navigation is working.
