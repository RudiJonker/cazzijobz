-- Function to get platform statistics (used in AdminScreen)
CREATE OR REPLACE FUNCTION get_platform_stats()
RETURNS TABLE(
    total_users BIGINT,
    total_workers BIGINT,
    total_employers BIGINT,
    total_jobs BIGINT,
    open_jobs BIGINT,
    completed_jobs BIGINT,
    total_applications BIGINT
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        (SELECT COUNT(*) FROM profiles) as total_users,
        (SELECT COUNT(*) FROM profiles WHERE role = 'worker') as total_workers,
        (SELECT COUNT(*) FROM profiles WHERE role = 'employer') as total_employers,
        (SELECT COUNT(*) FROM jobs) as total_jobs,
        (SELECT COUNT(*) FROM jobs WHERE status = 'open') as open_jobs,
        (SELECT COUNT(*) FROM jobs WHERE status = 'completed') as completed_jobs,
        (SELECT COUNT(*) FROM applications) as total_applications;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to get recent user signups (admin dashboard)
CREATE OR REPLACE FUNCTION get_recent_signups(days_back INTEGER DEFAULT 7)
RETURNS TABLE(
    user_id UUID,
    email TEXT,
    role TEXT,
    full_name TEXT,
    created_at TIMESTAMP WITH TIME ZONE,
    is_profile_complete BOOLEAN
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        p.id,
        p.email,
        p.role,
        p.full_name,
        p.created_at,
        p.is_profile_complete
    FROM profiles p
    WHERE p.created_at >= (NOW() - (days_back || ' days')::INTERVAL)
    ORDER BY p.created_at DESC;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to get job statistics by category
CREATE OR REPLACE FUNCTION get_job_stats_by_category()
RETURNS TABLE(
    category TEXT,
    total_jobs BIGINT,
    open_jobs BIGINT,
    avg_budget DECIMAL(10,2),
    total_applications BIGINT
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        j.category,
        COUNT(j.id) as total_jobs,
        COUNT(j.id) FILTER (WHERE j.status = 'open') as open_jobs,
        AVG(j.budget) as avg_budget,
        COUNT(a.id) as total_applications
    FROM jobs j
    LEFT JOIN applications a ON j.id = a.job_id
    GROUP BY j.category
    ORDER BY total_jobs DESC;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
